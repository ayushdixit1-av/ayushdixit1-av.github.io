class DistributedWorkerPool:
    def process_with_backoff(self, task: TaskEnvelope) -> TaskResult:
        try:
            self.heartbeat.pulse(worker_id=self.id, state="BUSY")
            return self.registry[task.fn].invoke(*task.args)
        except Exception as exc:
            if task.attempt >= task.max_retries:
                return self.redis.push_dead_letter_queue(task, reason=str(exc))
            delay = 2 ** task.attempt + jitter()
            return self.redis.schedule_retry(task.increment(), delay)
