class WAFTransformer:
    def __init__(self):
        self.encoder = TransformerEncoder(
            num_layers=6,
            num_heads=8,
            hidden_dim=512
        )
        self.classifier = MLPHead(
            num_classes=15
        )

    def detect_threat(self, request):
        tokens = self.tokenize(request)
        embeddings = self.encoder(tokens)
        threat_score = self.classifier(
            embeddings["[CLS]"]
        )
        return threat_score # 0.985
