class DistributedWorkerPool:
    def execute_with_backoff(self, job: JobEnvelope) -> JobResult:
        try:
            self.heartbeat.emit(worker_id=self.id, status="PROCESSING")
            return self.handlers[job.task].run(*job.args)
        except Exception as err:
            if job.attempt >= job.max_retries:
                return self.redis.move_to_dlq(job, error=str(err))
            backoff = (2 ** job.attempt) + uniform_jitter()
            return self.redis.enqueue_delayed(job.next_attempt(), backoff)
